<script>
const noBtn = document.getElementById('no');
const yesBtn = document.getElementById('yes');
const ask = document.getElementById('ask');
const success = document.getElementById('success');

let noScale = 1;
let yesScale = 1;

function shrinkNo(e) {
  if (e) e.preventDefault();
  noScale = Math.max(0.4, +(noScale - 0.15).toFixed(2));
  yesScale = Math.min(1.7, +(yesScale + 0.15).toFixed(2));
  noBtn.style.transform = 'scale(' + noScale + ')';
  yesBtn.style.transform = 'scale(' + yesScale + ')';
}

noBtn.addEventListener('click', shrinkNo);
noBtn.addEventListener('touchstart', shrinkNo);

yesBtn.addEventListener('click', () => {
  ask.style.display = 'none';
  success.style.display = 'block';
  for (let i = 0; i < 30; i++) {
    const h = document.createElement('div');
    h.textContent = ['💖','💕','💗','💘','❤️'][Math.floor(Math.random()*5)];
    h.style.position = 'absolute';
    h.style.left = Math.random()*100 + 'vw';
    h.style.top = '-2rem';
    h.style.fontSize = (1 + Math.random()*2) + 'rem';
    h.style.animation = 'fall ' + (3 + Math.random()*3) + 's linear forwards';
    document.getElementById('hearts').appendChild(h);
  }
});

const style = document.createElement('style');
style.textContent = '@keyframes fall { to { transform: translateY(110vh) rotate(360deg); opacity: 0.6; } }';
document.head.appendChild(style);
</script>
